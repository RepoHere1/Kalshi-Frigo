async function fetchRollData() { try { const response = await fetch('/api/roll-data'); // Adjust endpoint as necessary if (!response.ok) throw new Error('Network response was not ok' + response.statusText); const rollData = await response.json(); return rollData; } catch (error) { console.error('Error fetching roll data:', error); return []; } } fetchRollData().then((data) => { // Logic to render the roll card and update with data if (data.length > 0) { // Map data to update the Chart.js with roll data new Chart(document.getElementById('pnlChart'), { type: 'line', data: { labels: data.map(d => d.timestamp), datasets: [{ label: 'Roll Data', data: data.map(d => d.value), borderColor: '#3fb950', backgroundColor: 'transparent', tension: 0.3, }] }, options: { responsive: true, maintainAspectRatio: false, } }); } else { console.log('No roll data available.'); } });
